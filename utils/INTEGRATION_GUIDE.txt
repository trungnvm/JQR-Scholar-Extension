"""
Integration Guide: Impact Factor Collector → Browser Extension
================================================================

This guide shows how to collect impact factor data and integrate it
into the Scholar Extension for displaying journal quality indicators.

WORKFLOW OVERVIEW
-----------------

1. Prepare ISSN list from your target journals
2. Run Impact Factor Collector to gather data
3. Export to JavaScript format
4. Include in browser extension
5. Use data in extension UI


STEP-BY-STEP INTEGRATION
-------------------------

Step 1: Prepare Your ISSN List
-------------------------------

Option A: From existing CSV file
   import pandas as pd
   df = pd.read_csv('data/ISSNs.csv')
   issn_list = df['ISSN'].tolist()

Option B: From journal names (need to fetch ISSNs first)
   from journalListAggregator import JournalAggregator
   aggregator = JournalAggregator(email="your-email@example.com")
   df = aggregator.fetch_by_field("computer science", limit=1000)
   issn_list = [issn for issns in df['issn'] for issn in issns.split(',')]

Option C: Manual list
   issn_list = [
       "0028-0836",  # Nature
       "0036-8075",  # Science
       "0140-6736",  # The Lancet
   ]


Step 2: Collect Impact Factor Data
-----------------------------------

from impactFactorCollector import ImpactFactorCollector

# Initialize with SJR data
collector = ImpactFactorCollector(
    sjr_csv_path="data/raw/scimagojr.csv",
    email="your-email@example.com"
)

# Batch collect data
print(f"Processing {len(issn_list)} journals...")
results = collector.batch_collect(issn_list)

# Check results
successful = sum(1 for d in results.values() if d['if_value'] is not None)
print(f"Successfully collected data for {successful}/{len(issn_list)} journals")


Step 3: Export to JavaScript Format
------------------------------------

# Define output location in extension directory
output_file = "../extension/data/impact_factors.js"

# Export to JavaScript
collector.export_to_js(results, output_file)

print(f"Exported to: {output_file}")

# Optional: Also export to CSV for analysis
collector.export_to_csv(results, "data/processed/impact_factors.csv")


Step 4: Include in Browser Extension

Step 5: Use Data in Extension Code
-----------------------------------

// Check if data is loaded
if (typeof sfc !== 'undefined' && sfc.impactFactors) {
    console.log('Impact factor data loaded:',
                Object.keys(sfc.impactFactors).length, 'journals');
}

// Function to get impact factor for an ISSN
function getImpactFactor(issn) {
    // Normalize ISSN (remove hyphens)
    const normalizedISSN = issn.replace(/-/g, '');

    // Check both formats
    let data = sfc.impactFactors[issn] || sfc.impactFactors[normalizedISSN];

    if (data) {
        return {
            value: data.value,
            quartile: data.quartile,
            source: data.source,
            year: data.year
        };
    }

    return null;
}

// Function to display journal quality indicator
function displayQualityIndicator(issn, containerElement) {
    const ifData = getImpactFactor(issn);

    if (!ifData) {
        return; // No data available
    }

    // Create quality badge
    const badge = document.createElement('div');
    badge.className = 'quality-badge';

    // Color code by quartile
    const colors = {
        'Q1': '#28a745',  // Green
        'Q2': '#ffc107',  // Yellow
        'Q3': '#fd7e14',  // Orange
        'Q4': '#dc3545'   // Red
    };

    badge.style.backgroundColor = colors[ifData.quartile] || '#6c757d';
    badge.innerHTML = `
        <span class="quartile">${ifData.quartile}</span>
        <span class="if-value">IF: ${ifData.value.toFixed(2)}</span>
    `;

    containerElement.appendChild(badge);
}

// Example: Add to search results
document.querySelectorAll('.gs_ri').forEach(result => {
    const issnElement = result.querySelector('[data-issn]');
    if (issnElement) {
        const issn = issnElement.getAttribute('data-issn');
        displayQualityIndicator(issn, result);
    }
});


COMPLETE INTEGRATION SCRIPT
----------------------------

Save as: prepare_extension_data.py

#!/usr/bin/env python3
\"\"\"
Prepare impact factor data for browser extension.
\"\"\"

import pandas as pd
from pathlib import Path
from impactFactorCollector import ImpactFactorCollector

def main():
    print("=" * 60)
    print("Preparing Impact Factor Data for Extension")
    print("=" * 60)

    # Configuration
    SJR_CSV = "data/raw/scimagojr.csv"
    ISSN_FILE = "data/ISSNs.csv"
    OUTPUT_JS = "../extension/data/impact_factors.js"
    OUTPUT_CSV = "data/processed/impact_factors.csv"
    EMAIL = "your-email@example.com"

    # Load ISSN list
    print(f"\n1. Loading ISSN list from: {ISSN_FILE}")
    df = pd.read_csv(ISSN_FILE)
    issn_list = df['ISSN'].dropna().unique().tolist()
    print(f"   Found {len(issn_list)} unique ISSNs")

    # Initialize collector
    print(f"\n2. Initializing collector with SJR data: {SJR_CSV}")
    collector = ImpactFactorCollector(
        sjr_csv_path=SJR_CSV,
        email=EMAIL
    )

    # Collect data
    print(f"\n3. Collecting impact factor data...")
    results = collector.batch_collect(issn_list)

    # Statistics
    sources = {}
    for data in results.values():
        source = data['source']
        sources[source] = sources.get(source, 0) + 1

    print(f"\n4. Collection complete!")
    print(f"   Total journals: {len(results)}")
    print(f"   Data sources:")
    for source, count in sources.items():
        print(f"     - {source}: {count}")

    # Export to JavaScript
    print(f"\n5. Exporting to JavaScript: {OUTPUT_JS}")
    Path(OUTPUT_JS).parent.mkdir(parents=True, exist_ok=True)
    collector.export_to_js(results, OUTPUT_JS)

    # Export to CSV
    print(f"\n6. Exporting to CSV: {OUTPUT_CSV}")
    Path(OUTPUT_CSV).parent.mkdir(parents=True, exist_ok=True)
    collector.export_to_csv(results, OUTPUT_CSV)

    print("\n" + "=" * 60)
    print("✓ Data preparation complete!")
    print("=" * 60)
    print(f"\nNext steps:")
    print(f"1. Verify {OUTPUT_JS} exists")
    print(f"2. Include in extension manifest.json")
    print(f"3. Use sfc.impactFactors in your extension code")
    print(f"4. Test with: chrome://extensions")

if __name__ == "__main__":
    main()

   console.log(Object.keys(sfc.impactFactors).length);

   console.log(sfc.impactFactors['0028-0836']);


UPDATING DATA
-------------

To refresh impact factor data:

1. Download latest SJR CSV from scimagojr.com
2. Run prepare_extension_data.py again
3. Reload extension in browser
4. Verify data is updated


TROUBLESHOOTING
---------------

Issue: sfc is undefined
Solution: Ensure impact_factors.js is loaded before content.js

Issue: No data for known journal
Solution: Check ISSN format (with/without hyphen), verify in CSV

Issue: Extension won't load
Solution: Check manifest.json syntax, verify file paths

Issue: Data file too large
Solution: Filter to only needed journals, remove unused fields


PERFORMANCE OPTIMIZATION
-------------------------

1. Minimize data file:
   - Only include journals you need
   - Remove unused fields from export
   - Consider compression

2. Lazy loading:
   - Load data only when needed
   - Cache in extension storage

3. Update frequency:
   - Update quarterly (SJR updates annually)
   - Don't update too frequently


BEST PRACTICES
--------------

1. Always include data source and year in UI
2. Handle missing data gracefully
3. Provide fallback for unknown journals
4. Cache impact factor lookups
5. Update data regularly but not excessively
6. Test with various journal ISSNs
7. Log errors for debugging
8. Validate data before deployment


SUPPORT & MAINTENANCE
----------------------

Regular maintenance tasks:

1. Monthly: Check for SJR data updates
2. Quarterly: Refresh impact factor data
3. Annually: Major update with new SJR year
4. As needed: Add new journals from user requests

For issues:
- Check logs in browser console
- Verify data file integrity
- Test with known good ISSNs
